cmake_minimum_required(VERSION 3.20)
project(openbor_libretro C RC)
if(NOT WIN32)
  message(FATAL_ERROR "This initial backend uses Windows fibers (Windows x64).")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The supported target is Windows x64 (UCRT64).")
endif()
set(CMAKE_C_STANDARD 99)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
include(Sources.cmake)
add_library(openbor_engine SHARED core.c utf8_io.c ${ROOT}/openbor.c ${ROOT}/openborscript.c ${ROOT}/sdl/threads.c ${SOURCES})
set_source_files_properties(utf8_io.c PROPERTIES COMPILE_DEFINITIONS BOR_UTF8_IMPLEMENTATION=1)
set_target_properties(openbor_engine PROPERTIES PREFIX "")
target_include_directories(openbor_engine PRIVATE . ${ROOT} ${ROOT}/sdl)
foreach(DIR source source/adpcmlib source/gamelib source/preprocessorlib source/ramlib source/randlib source/scriptlib source/openborscript source/pnglib source/gfxlib source/webmlib)
 target_include_directories(openbor_engine PRIVATE ${ROOT}/${DIR})
endforeach()
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED sdl2 libpng vorbisfile)
target_include_directories(openbor_engine PRIVATE ${DEPS_INCLUDE_DIRS})
target_link_directories(openbor_engine PRIVATE ${DEPS_STATIC_LIBRARY_DIRS})
target_compile_definitions(openbor_engine PRIVATE LIBRETRO=1 WIN=1 SDL=1 SDL_MAIN_HANDLED=1 SKIP_CODE=1 NO_RAM_DEBUGGER=1 OV_EXCLUDE_STATIC_CALLBACKS=1)
target_compile_options(openbor_engine PRIVATE -fsigned-char -fcommon -Wno-incompatible-pointer-types -Wno-int-conversion -include ${CMAKE_CURRENT_SOURCE_DIR}/utf8_io.h)
target_link_options(openbor_engine PRIVATE -static -static-libgcc)
target_link_libraries(openbor_engine PRIVATE ${DEPS_STATIC_LIBRARIES} psapi m)
# Embed a fresh engine image in the single public core DLL. OpenBOR contains
# process-lifetime globals; loading a private image per session avoids sharing
# those globals between resets, content switches or simultaneous EmuVR screens.
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/engine.rc" CONTENT "1 RCDATA \"$<TARGET_FILE:openbor_engine>\"\n")
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/engine.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/openbor_engine.dll")
add_library(openbor_libretro SHARED session.c "${CMAKE_CURRENT_BINARY_DIR}/engine.rc")
add_dependencies(openbor_libretro openbor_engine)
set_target_properties(openbor_libretro PROPERTIES PREFIX "")
target_link_options(openbor_libretro PRIVATE -static -static-libgcc)

include(CTest)
if(BUILD_TESTING)
  add_executable(smoke tests/smoke.c)
  target_include_directories(smoke PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_options(smoke PRIVATE -municode -static)
  # Isolate the actual block controller so tests can replace animation/output
  # side effects without linking the full game and script interpreter.
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/openbor.c")
  file(READ "${ROOT}/openbor.c" ENGINE_TEXT)
  string(FIND "${ENGINE_TEXT}" "void common_block()" BLOCK_START)
  string(FIND "${ENGINE_TEXT}" "void common_charge()" BLOCK_END)
  if(BLOCK_START LESS 0 OR BLOCK_END LESS BLOCK_START)
    message(FATAL_ERROR "Cannot locate block controller for regression test")
  endif()
  math(EXPR BLOCK_LENGTH "${BLOCK_END} - ${BLOCK_START}")
  string(SUBSTRING "${ENGINE_TEXT}" ${BLOCK_START} ${BLOCK_LENGTH} BLOCK_TEXT)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/block_controller.inc" "${BLOCK_TEXT}")
  string(FIND "${ENGINE_TEXT}" "void dropweapon(int flag)" WEAPON_START)
  string(FIND "${ENGINE_TEXT}" "int player_takedamage(entity *other, s_attack *attack, int fall_flag, s_defense* defense_object)" WEAPON_END)
  if(WEAPON_START LESS 0 OR WEAPON_END LESS WEAPON_START)
    message(FATAL_ERROR "Cannot locate weapon controller for regression test")
  endif()
  math(EXPR WEAPON_LENGTH "${WEAPON_END} - ${WEAPON_START}")
  string(SUBSTRING "${ENGINE_TEXT}" ${WEAPON_START} ${WEAPON_LENGTH} WEAPON_TEXT)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/weapon_controller.inc" "${WEAPON_TEXT}")
  foreach(TEST mixer_regression death_regression block_regression selection_regression weapon_regression)
    add_executable(${TEST} tests/${TEST}.c)
    target_include_directories(${TEST} PRIVATE ${CMAKE_CURRENT_BINARY_DIR} $<TARGET_PROPERTY:openbor_engine,INCLUDE_DIRECTORIES>)
    target_compile_definitions(${TEST} PRIVATE $<TARGET_PROPERTY:openbor_engine,COMPILE_DEFINITIONS>)
    target_compile_options(${TEST} PRIVATE -O2 -fwhole-program -ffunction-sections -fdata-sections -Wno-incompatible-pointer-types -Wno-int-conversion)
    target_link_options(${TEST} PRIVATE -Wl,--gc-sections -static)
    add_test(NAME ${TEST} COMMAND ${TEST})
  endforeach()
endif()
